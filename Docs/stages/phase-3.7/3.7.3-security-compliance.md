// convex/security/authentication.ts
import { mutation, query } from "../_generated/server";

export const enhancedLogin = mutation({
  args: {
    identifier: v.string(), // email or phone

    otp: v.optional(v.string()),
    deviceFingerprint: v.string(),
    ipAddress: v.string(),
    userAgent: v.string(),

      city: v.string(),
      latitude: v.optional(v.number()),

  },
  handler: async (ctx, args) => {
    // Log authentication attempt
    const attemptId = await ctx.db.insert("authentication_attempts", {
      identifier: args.identifier,
      attemptType: "login",
      success: false, // Will update if successful
      ipAddress: args.ipAddress,
      userAgent: args.userAgent,
      location: args.location,
      timestamp: Date.now(),
    });

    try {
      // Find user
      const user = await ctx.db
        .query("users")
        .withIndex("by_email", (q) => q.eq("email", args.identifier))
        .first() ||
        await ctx.db
        .query("users")
        .withIndex("by_phone", (q) => q.eq("phone", args.identifier))
        .first();

      if (!user) {
        throw new Error("User not found");
      }

      // Check if account is locked due to failed attempts
      const recentFailures = await ctx.db
        .query("authentication_attempts")
        .withIndex("by_identifier", (q) => q.eq("identifier", args.identifier))
        .filter((q) => 
          q.and(
            q.eq(q.field("success"), false),
            q.gt(q.field("timestamp"), Date.now() - 15 * 60 * 1000) // 15 minutes
          )
        )
        .collect();

      if (recentFailures.length >= 5) {
        // Create security incident
        await ctx.db.insert("security_incidents", {
          type: "brute_force",
          severity: "medium",
          description: `Multiple failed login attempts for ${args.identifier}`,
          affectedUserId: user._id,
          sourceIp: args.ipAddress,
          userAgent: args.userAgent,
          details: { attemptCount: recentFailures.length },
          status: "open",
          detectedAt: Date.now(),
        });
        
        throw new Error("Account temporarily locked due to multiple failed attempts");
      }

      // Verify OTP (simplified - integrate with existing OTP system)
      if (args.otp) {
        const otpValid = await verifyOTP(ctx, user._id, args.otp);
        if (!otpValid) {
          throw new Error("Invalid OTP");
        }
      }

      // Check if device is trusted
      const trustedDevice = await ctx.db
        .query("trusted_devices")
        .withIndex("by_fingerprint", (q) => q.eq("deviceFingerprint", args.deviceFingerprint))
        .filter((q) => 
          q.and(
            q.eq(q.field("userId"), user._id),
            q.eq(q.field("isActive"), true)
          )
        )
        .first();

      // Risk assessment
      const riskScore = await calculateLoginRisk(ctx, user._id, args);
      
      if (riskScore > 70 && !trustedDevice) {
        // Require additional verification
        return {
          success: false,
          requiresAdditionalVerification: true,
          verificationMethods: ["sms", "email"],
          riskScore,
        };
      }

      // Check for suspicious patterns
      await detectAnomalousLogin(ctx, user._id, args);

      // Register/update device
      if (!trustedDevice) {
        await ctx.db.insert("trusted_devices", {
          userId: user._id,
          deviceFingerprint: args.deviceFingerprint,
          deviceName: extractDeviceName(args.userAgent),
          deviceType: extractDeviceType(args.userAgent),
          browser: extractBrowser(args.userAgent),
          ipAddress: args.ipAddress,
          location: args.location,
          isActive: true,
          lastSeen: Date.now(),
          registeredAt: Date.now(),
        });
      } else {
        await ctx.db.patch(trustedDevice._id, {
          lastSeen: Date.now(),
          ipAddress: args.ipAddress,
          location: args.location,
        });
      }

      // Update successful attempt
      await ctx.db.patch(attemptId, {
        success: true,
        userId: user._id,
      });

      // Create session (integrate with existing session management)
      const sessionToken = await createSecureSession(ctx, user._id, args);

      return {
        success: true,
        user: {
          id: user._id,
          email: user.email,
          role: user.role,
        },
        sessionToken,
        requiresTwoFactor: await requiresTwoFactor(ctx, user._id),
      };

    } catch (error) {
      // Update failed attempt with reason
      await ctx.db.patch(attemptId, {
        failureReason: error.message,
      });
      throw error;
    }
  },
});

export const setupTwoFactor = mutation({
  args: {
    userId: v.id("users"),
    method: v.union(v.literal("sms"), v.literal("email"), v.literal("app")),
  },
  handler: async (ctx, args) => {
    // Generate secret for authenticator app
    const secret = crypto.randomBytes(32).toString('hex');
    
    // Generate backup codes
    const backupCodes = Array.from({ length: 10 }, () => 
      crypto.randomBytes(4).toString('hex').toUpperCase()
    );

    // Encrypt sensitive data
    const encryptedSecret = encrypt(secret);
    const encryptedBackupCodes = backupCodes.map(code => encrypt(code));

    // Store 2FA settings
    const twoFactorId = await ctx.db.insert("two_factor_auth", {
      userId: args.userId,
      method: args.method,
      secret: encryptedSecret,
      backupCodes: encryptedBackupCodes,
      isEnabled: false, // Will be enabled after verification
      createdAt: Date.now(),
    });

    // Log the setup
    await ctx.db.insert("user_activity_logs", {
      userId: args.userId,
      action: "two_factor_setup_initiated",
      details: { method: args.method },
      ipAddress: "", // TODO: Get from context

    });

    return {
      success: true,
      secret: secret, // Return unencrypted for QR code generation

    };
  },

export const getSecurityMetrics = query({
  args: { 
    estateId: v.optional(v.id("estates")),
    timeRange: v.optional(v.object({
      start: v.number(),
      end: v.number(),
    }))
  },
  handler: async (ctx, args) => {

      start: Date.now() - (30 * 24 * 60 * 60 * 1000), // 30 days ago
      end: Date.now(),
    };

    // Get security incidents
    let incidentsQuery = ctx.db
      .query("security_incidents")
      .withIndex("by_detected_at", (q) => 
        q.gte("detectedAt", timeRange.start).lte("detectedAt", timeRange.end)
      );

    if (args.estateId) {
      incidentsQuery = incidentsQuery.filter((q) => 
        q.eq(q.field("affectedEstateId"), args.estateId)
      );
    }

    const incidents = await incidentsQuery.collect();

    // Get authentication attempts
    const authAttempts = await ctx.db
      .query("authentication_attempts")
      .withIndex("by_timestamp", (q) => 
        q.gte("timestamp", timeRange.start).lte("timestamp", timeRange.end)
      )
      .collect();

    // Calculate metrics
    const metrics = {
      totalIncidents: incidents.length,
      incidentsBySeverity: {
        low: incidents.filter(i => i.severity === "low").length,
        medium: incidents.filter(i => i.severity === "medium").length,
        high: incidents.filter(i => i.severity === "high").length,
        critical: incidents.filter(i => i.severity === "critical").length,
      },
      incidentsByType: incidents.reduce((acc, incident) => {
        acc[incident.type] = (acc[incident.type] || 0) + 1;
        return acc;
      }, {} as Record<string, number>),
      authenticationMetrics: {
        totalAttempts: authAttempts.length,
        successfulLogins: authAttempts.filter(a => a.success).length,
        failedAttempts: authAttempts.filter(a => !a.success).length,
        successRate: authAttempts.length > 0 ? 
          (authAttempts.filter(a => a.success).length / authAttempts.length) * 100 : 0,
      },
      topRiskIPs: getTopRiskIPs(authAttempts),
      securityScore: calculateOverallSecurityScore(incidents, authAttempts),
    };

    return metrics;
  },
});

function getTopRiskIPs(authAttempts: any[]): Array<{ip: string, attempts: number, successRate: number}> {
  const ipStats = authAttempts.reduce((acc, attempt) => {
    if (!acc[attempt.ipAddress]) {
      acc[attempt.ipAddress] = { total: 0, successful: 0 };
    }
    acc[attempt.ipAddress].total++;
    if (attempt.success) acc[attempt.ipAddress].successful++;
    return acc;
  }, {} as Record<string, {total: number, successful: number}>);

  return Object.entries(ipStats)
    .map(([ip, stats]) => ({
      ip,
      attempts: stats.total,
      successRate: (stats.successful / stats.total) * 100,
    }))
    .filter(stat => stat.attempts > 5 && stat.successRate < 50)
    .sort((a, b) => b.attempts - a.attempts)
    .slice(0, 10);
}

function calculateOverallSecurityScore(incidents: any[], authAttempts: any[]): number {
  let score = 100;
  
  // Deduct points for incidents
  score -= incidents.filter(i => i.severity === "critical").length * 20;
  score -= incidents.filter(i => i.severity === "high").length * 10;
  score -= incidents.filter(i => i.severity === "medium").length * 5;
  
  // Deduct points for failed authentication attempts
  const failureRate = authAttempts.length > 0 ? 
    (authAttempts.filter(a => !a.success).length / authAttempts.length) * 100 : 0;
  score -= failureRate * 0.5;
  
  return Math.max(score, 0);
}
