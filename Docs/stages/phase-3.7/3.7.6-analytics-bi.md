// convex/analytics/dataProcessor.ts
import { mutation, query, action } from "../_generated/server";
import { v } from "convex/values";

export const processVisitorAnalytics = mutation({
  args: {
    estateId: v.id("estates"),
    date: v.string(),
  },
  handler: async (ctx, args) => {
    const dateStart = new Date(args.date).getTime();
    const dateEnd = dateStart + (24 * 60 * 60 * 1000) - 1;

    // Get all visits for the date
    const visits = await ctx.db
      .query("visits")
      .withIndex("by_estate_date", (q) => 
        q.eq("estateId", args.estateId).gte("checkinTime", dateStart).lte("checkinTime", dateEnd)
      )
      .collect();

    // Calculate metrics
    const totalVisitors = visits.length;
    const walkInVisitors = visits.filter(v => !v.invitationId).length;
    const preRegisteredVisitors = visits.filter(v => v.invitationId).length;

    // Calculate average stay duration
    const completedVisits = visits.filter(v => v.checkoutTime);
    const averageStayDuration = completedVisits.length > 0 ? 
      completedVisits.reduce((sum, v) => sum + (v.checkoutTime! - v.checkinTime), 0) / completedVisits.length : 0;

    // Find peak hour
    const hourlyVisits = new Array(24).fill(0);
    visits.forEach(visit => {
      const hour = new Date(visit.checkinTime).getHours();
      hourlyVisits[hour]++;
    });
    const peakHour = hourlyVisits.indexOf(Math.max(...hourlyVisits));

    // Popular purposes
    const purposeCounts = visits.reduce((acc, visit) => {
      acc[visit.purpose] = (acc[visit.purpose] || 0) + 1;
      return acc;
    }, {} as Record<string, number>);

    const popularPurposes = Object.entries(purposeCounts)
      .map(([purpose, count]) => ({ purpose, count }))
      .sort((a, b) => b.count - a.count);

    // Get comparison data
    const previousDay = await getPreviousDayData(ctx, args.estateId, args.date);
    const previousWeek = await getPreviousWeekData(ctx, args.estateId, args.date);
    const previousMonth = await getPreviousMonthData(ctx, args.estateId, args.date);

    // Hourly breakdown
    const hourlyBreakdown = hourlyVisits.map((count, hour) => {
      const hourVisits = visits.filter(v => new Date(v.checkinTime).getHours() === hour);
      const avgDuration = hourVisits.length > 0 && hourVisits.filter(v => v.checkoutTime).length > 0 ?
        hourVisits.filter(v => v.checkoutTime).reduce((sum, v) => sum + (v.checkoutTime! - v.checkinTime), 0) / hourVisits.filter(v => v.checkoutTime).length : 0;
      
      return {
        hour,
        visitorCount: count,
        avgDuration,
      };
    });

    // Store analytics
    const analyticsId = await ctx.db.insert("visitor_analytics", {
      estateId: args.estateId,
      date: args.date,
      metrics: {
        totalVisitors,
        walkInVisitors,
        preRegisteredVisitors,
        averageStayDuration,
        peakHour: `${peakHour}:00`,
        popularPurposes,
        returningVisitors: 0, // TODO: Implement returning visitor logic
        newVisitors: totalVisitors, // TODO: Implement new visitor logic
      },
      hourlyBreakdown,
      comparisonData: {
        previousDay: previousDay?.metrics.totalVisitors || 0,
        previousWeek: previousWeek?.metrics.totalVisitors || 0,
        previousMonth: previousMonth?.metrics.totalVisitors || 0,
      },
    });

    return { success: true, analyticsId };
  },
});

export const processSecurityAnalytics = mutation({
  args: {
    estateId: v.id("estates"),
    date: v.string(),
  },
  handler: async (ctx, args) => {
    const dateStart = new Date(args.date).getTime();
    const dateEnd = dateStart + (24 * 60 * 60 * 1000) - 1;

    // Get security incidents
    const incidents = await ctx.db
      .query("security_incidents")
      .withIndex("by_detected_at", (q) => 
        q.gte("detectedAt", dateStart).lte("detectedAt", dateEnd)
      )
      .filter((q) => q.eq(q.field("affectedEstateId"), args.estateId))
      .collect();

    // Get authentication attempts
    const authAttempts = await ctx.db
      .query("authentication_attempts")
      .withIndex("by_timestamp", (q) => 
        q.gte("timestamp", dateStart).lte("timestamp", dateEnd)
      )
      .collect();

    // Get overstay alerts
    const visits = await ctx.db
      .query("visits")
      .withIndex("by_estate_date", (q) => 
        q.eq("estateId", args.estateId).gte("checkinTime", dateStart).lte("checkinTime", dateEnd)
      )
      .collect();

    const estate = await ctx.db.get(args.estateId);
    const overstayThreshold = estate?.settings.overstayThresholdHours || 8;
    const overstayAlerts = visits.filter(v => 
      v.status === "checked_in" && 
      (Date.now() - v.checkinTime) > (overstayThreshold * 60 * 60 * 1000)
    ).length;

    // Calculate metrics
    const incidentsBySeverity = incidents.reduce((acc, incident) => {
      acc[incident.severity] = (acc[incident.severity] || 0) + 1;
      return acc;
    }, { low: 0, medium: 0, high: 0, critical: 0 });

    const failedLoginAttempts = authAttempts.filter(a => !a.success).length;
    const successfulLogins = authAttempts.filter(a => a.success).length;

    // Response time calculations (placeholder - implement based on your incident response system)
    const responseTime = {
      average: 15, // minutes
      p95: 30,
    };

    // Get previous data for trends
    const previousDate = new Date(new Date(args.date).getTime() - (24 * 60 * 60 * 1000)).toISOString().split('T')[0];
    const previousData = await ctx.db
      .query("security_analytics")
      .withIndex("by_estate_date", (q) => q.eq("estateId", args.estateId).eq("date", previousDate))
      .first();

    const trends = {
      incidentTrend: previousData ? 
        ((incidents.length - previousData.metrics.totalIncidents) / Math.max(previousData.metrics.totalIncidents, 1)) * 100 : 0,
      loginTrend: previousData ? 
        ((successfulLogins - previousData.metrics.successfulLogins) / Math.max(previousData.metrics.successfulLogins, 1)) * 100 : 0,
      responseTimeTrend: previousData ? 
        ((responseTime.average - previousData.metrics.responseTime.average) / Math.max(previousData.metrics.responseTime.average, 1)) * 100 : 0,
    };

    const analyticsId = await ctx.db.insert("security_analytics", {
      estateId: args.estateId,
      date: args.date,
      metrics: {
        totalIncidents: incidents.length,
        incidentsBySeverity,
        failedLoginAttempts,
        successfulLogins,
        overstayAlerts,
        responseTime,
      },
      trends,
    });

    return { success: true, analyticsId };
  },
});

export const runDailyAnalytics = action({
  args: { estateId: v.id("estates") },
  handler: async (ctx, args) => {
    const today = new Date().toISOString().split('T')[0];
    
    try {
      // Process visitor analytics
      await ctx.runMutation("analytics/dataProcessor.processVisitorAnalytics", {
        estateId: args.estateId,
        date: today,
      });

      // Process security analytics
      await ctx.runMutation("analytics/dataProcessor.processSecurityAnalytics", {
        estateId: args.estateId,
        date: today,
      });

      // Process operational analytics
      await ctx.runMutation("analytics/dataProcessor.processOperationalAnalytics", {
        estateId: args.estateId,
        date: today,
      });

      return { success: true, date: today };
    } catch (error) {
      console.error("Daily analytics processing failed:", error);
      throw error;
    }
  },
});

interface MetricCardProps {

  value: string | number;
  change: number;
  icon: any;
  color: 'blue' | 'green' | 'purple' | 'orange';
}

function MetricCard({ title, value, change, icon: Icon, color }: MetricCardProps) {
  const colorClasses = {
    blue: 'text-blue-600',
    green: 'text-green-600',
    purple: 'text-purple-600',
    orange: 'text-orange-600',
  };

  return (
    <Card>
      <CardContent className="pt-6">
        <div className="flex items-center justify-between">
          <div>
            <p className="text-sm font-medium text-gray-600">{title}</p>

              {change >= 0 ? (
                <TrendingUp className="w-4 h-4 text-green-500 mr-1" />
              ) : (
                <TrendingDown className="w-4 h-4 text-red-500 mr-1" />
              )}
              <span className={`text-sm ${change >= 0 ? 'text-green-500' : 'text-red-500'}`}>
                {Math.abs(change).toFixed(1)}%
              </span>
            </div>
          </div>
          <Icon className={`w-8 h-8 ${colorClasses[color]}`} />
        </div>
      </CardContent>
    </Card>
  );
}

function PeakHoursHeatmap({ data }: { data: any[] }) {
  const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];

  return (

        <div key={day} className="text-xs font-medium mb-1">{day}</div>
      ))}
      {data.map((hour, index) => (
        <div
          key={index}
          className={`w-3 h-3 rounded ${getHeatmapColor(hour.intensity)}`}
          title={`${hour.hour}:00 - ${hour.visitors} visitors`}
        />
      ))}
    </div>
  );
}

function getHeatmapColor(intensity: number): string {
  if (intensity === 0) return 'bg-gray-100';
  if (intensity < 0.25) return 'bg-blue-200';
  if (intensity < 0.5) return 'bg-blue-400';
  if (intensity < 0.75) return 'bg-blue-600';
  return 'bg-blue-800';
}

// components/analytics/ReportBuilder.tsx
"use client";

import { useState } from 'react';
import { useMutation, useQuery } from 'convex/react';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';

  Table

interface ReportBuilderProps {

  onClose: () => void;
}

export function ReportBuilder({ estateId, onClose }: ReportBuilderProps) {
  const [reportConfig, setReportConfig] = useState({

  const generateReport = useMutation(api.analytics.reports.generateReport);

    { id: 'visitor_count', label: 'Visitor Count', category: 'Visitors' },
    { id: 'average_stay_time', label: 'Average Stay Time', category: 'Visitors' },
    { id: 'peak_hours', label: 'Peak Hours', category: 'Visitors' },
    { id: 'purpose_distribution', label: 'Purpose Distribution', category: 'Visitors' },
    { id: 'security_incidents', label: 'Security Incidents', category: 'Security' },
    { id: 'failed_logins', label: 'Failed Login Attempts', category: 'Security' },
    { id: 'response_time', label: 'Response Time', category: 'Security' },
    { id: 'system_uptime', label: 'System Uptime', category: 'Operations' },
    { id: 'user_adoption', label: 'User Adoption Rate', category: 'Operations' },
    { id: 'efficiency_score', label: 'Efficiency Score', category: 'Operations' },
  ];

  const handleMetricToggle = (metricId: string) => {
    setReportConfig(prev => ({
      ...prev,
      metrics: prev.metrics.includes(metricId)
        ? prev.metrics.filter(m => m !== metricId)
        : [...prev.metrics, metricId]
    }));
  };

  const handleVisualizationToggle = (vizType: string) => {
    setReportConfig(prev => ({
      ...prev,
      visualizations: prev.visualizations.includes(vizType)
        ? prev.visualizations.filter(v => v !== vizType)
        : [...prev.visualizations, vizType]
    }));
  };

  const handleSaveReport = async () => {
    try {
      const reportId = await createReport({
        estateId,
        name: reportConfig.name,
        description: reportConfig.description,
        reportType: reportConfig.reportType as any,
        configuration: {
          dateRange: {
            start: reportConfig.dateRange.from.getTime(),
            end: reportConfig.dateRange.to.getTime(),
          },
          metrics: reportConfig.metrics,
          filters: reportConfig.filters,
          visualizations: reportConfig.visualizations as any,
        },
        schedule: reportConfig.schedule.isActive ? reportConfig.schedule : undefined,
      });

      onClose();
    } catch (error) {
      console.error('Failed to save report:', error);
    }
  };

  const handleGenerateReport = async () => {
    setGenerating(true);
    try {
      const result = await generateReport({
        estateId,
        configuration: {
          dateRange: {
            start: reportConfig.dateRange.from.getTime(),
            end: reportConfig.dateRange.to.getTime(),
          },
          metrics: reportConfig.metrics,
          filters: reportConfig.filters,
          visualizations: reportConfig.visualizations as any,
        },
        format: 'pdf',
      });

      // Download the generated report
      if (result.downloadUrl) {
        window.open(result.downloadUrl, '_blank');
      }
    } catch (error) {
      console.error('Failed to generate report:', error);
    } finally {
      setGenerating(false);
    }
  };

  return (
    <div className="space-y-6">
      <Card>
        <CardHeader>
          <CardTitle>Custom Report Builder</CardTitle>
          <CardDescription>
            Create custom reports with specific metrics and visualizations
          </CardDescription>
        </CardHeader>
        <CardContent className="space-y-6">
          {/* Basic Information */}
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            <div>
              <Label htmlFor="reportName">Report Name</Label>
              <Input
                id="reportName"
                value={reportConfig.name}
                onChange={(e) => setReportConfig(prev => ({ ...prev, name: e.target.value }))}
                placeholder="Monthly Visitor Analysis"
              />
            </div>
            <div>
              <Label htmlFor="reportType">Report Type</Label>
              <Select
                value={reportConfig.reportType}
                onValueChange={(value) => setReportConfig(prev => ({ ...prev, reportType: value }))}
              >
                <SelectTrigger>
                  <SelectValue placeholder="Select report type" />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="visitor_summary">Visitor Summary</SelectItem>
                  <SelectItem value="security_audit">Security Audit</SelectItem>
                  <SelectItem value="operational_efficiency">Operational Efficiency</SelectItem>
                  <SelectItem value="compliance_report">Compliance Report</SelectItem>
                  <SelectItem value="custom_query">Custom Query</SelectItem>
                </SelectContent>
              </Select>
            </div>
          </div>

          <div>
            <Label htmlFor="description">Description</Label>
            <Textarea
              id="description"
              value={reportConfig.description}
              onChange={(e) => setReportConfig(prev => ({ ...prev, description: e.target.value }))}
              placeholder="Describe what this report covers..."
            />
          </div>

          {/* Date Range */}
          <div>
            <Label>Date Range</Label>
            <DatePickerWithRange
              date={reportConfig.dateRange}
              onDateChange={(range) => setReportConfig(prev => ({ ...prev, dateRange: range }))}
            />
          </div>

          {/* Metrics Selection */}
          <div>
            <Label>Metrics to Include</Label>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mt-2">
              {['Visitors', 'Security', 'Operations'].map(category => (
                <div key={category}>
                  <h4 className="font-semibold text-sm mb-2">{category}</h4>
                  <div className="space-y-2">
                    {availableMetrics
                      .filter(metric => metric.category === category)
                      .map(metric => (
                        <div key={metric.id} className="flex items-center space-x-2">
                          <Checkbox
                            id={metric.id}
                            checked={reportConfig.metrics.includes(metric.id)}
                            onCheckedChange={() => handleMetricToggle(metric.id)}
                          />
                          <label htmlFor={metric.id} className="text-sm">
                            {metric.label}
                          </label>
                        </div>
                      ))}
                  </div>
                </div>
              ))}
            </div>
          </div>

          {/* Visualization Types */}
          <div>
            <Label>Visualizations</Label>
            <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mt-2">
              {[
                { id: 'line_chart', label: 'Line Chart', icon: LineChart },
                { id: 'bar_chart', label: 'Bar Chart', icon: BarChart3 },
                { id: 'pie_chart', label: 'Pie Chart', icon: PieChart },
                { id: 'table', label: 'Table', icon: Table },
              ].map(viz => (
                <div key={viz.id} className="flex items-center space-x-2">
                  <Checkbox
                    id={viz.id}
                    checked={reportConfig.visualizations.includes(viz.id)}
                    onCheckedChange={() => handleVisualizationToggle(viz.id)}
                  />
                  <label htmlFor={viz.id} className="text-sm flex items-center gap-1">
                    <viz.icon className="w-4 h-4" />
                    {viz.label}
                  </label>
                </div>
              ))}
            </div>
          </div>

            <Button variant="outline" onClick={handleSaveReport}>
              <FileText className="w-4 h-4 mr-2" />

            </Button>
            <Button variant="outline" onClick={onClose}>
              Cancel
            </Button>
          </div>
        </CardContent>
      </Card>
    </div>
  );
}
