```typescript
// convex/middleware/estateIsolation.ts
import { query, mutation } from "../_generated/server";

// Middleware to ensure all operations are estate-scoped
export const withEstateContext = (handler: any) => {
  return async (ctx: any, args: any) => {
    const { estateId, userId, ...otherArgs } = args;

    if (!estateId) {
      throw new Error("Estate context required for this operation");
    }

    // Verify user has access to this estate
    const userEstateAccess = await ctx.db
      .query("estate_users")
      .withIndex("by_user", (q) => q.eq("userId", userId))
      .filter((q) => 
        q.and(
          q.eq(q.field("estateId"), estateId),
          q.eq(q.field("isActive"), true)
        )
      )
      .first();

    if (!userEstateAccess) {
      throw new Error("User does not have access to this estate");
    }

    // Add estate context to args
    const contextualArgs = {
      ...otherArgs,
      estateId,
      userId,
      userRole: userEstateAccess.role,
      userPermissions: userEstateAccess.permissions,
    };

    return handler(ctx, contextualArgs);
  };
};

// Helper to get user's accessible estates
export const getUserEstates = query({
  args: { userId: v.id("users") },
  handler: async (ctx, args) => {
    const estateAccess = await ctx.db
      .query("estate_users")
      .withIndex("by_user", (q) => q.eq("userId", args.userId))
      .filter((q) => q.eq(q.field("isActive"), true))
      .collect();

    const estates = await Promise.all(
      estateAccess.map(async (access) => {
        const estate = await ctx.db.get(access.estateId);
        return {
          ...estate,
          userRole: access.role,
          userPermissions: access.permissions,
        };
      })
    );

    return estates.filter(Boolean);
  },
});
```

```typescript
import { mutation, query } from "./_generated/server";
import { v } from "convex/values";

export const createEstate = mutation({
  args: {
    name: v.string(),
    address: v.string(),

      email: v.string(),
      phone: v.string(),
      website: v.optional(v.string()),

      name: args.name,
      address: args.address,
      contact: args.contact,
      billing: {
        plan: args.plan,
        maxResidents: args.plan === "basic" ? 50 : args.plan === "premium" ? 200 : 1000,
        maxGuests: args.plan === "basic" ? 500 : args.plan === "premium" ? 2000 : 10000,
        billingCycle: "monthly",
        nextBillingDate: Date.now() + (30 * 24 * 60 * 60 * 1000),
        isActive: true,
      },
      settings: {
        overstayThresholdHours: 8,
        requirePhotoUpload: false,
        allowWalkIns: true,
        workingHours: { start: "08:00", end: "18:00" },
        features: {
          smsNotifications: args.plan !== "basic",
          emailNotifications: true,
          qrCodeEntry: true,
          biometricEntry: args.plan === "enterprise",
          vehicleTracking: args.plan !== "basic",
          guestPreRegistration: true,
          reportGeneration: true,
          analyticsAccess: args.plan !== "basic",
        },
        security: {
          sessionTimeout: 30 * 60 * 1000, // 30 minutes
          passwordPolicy: {
            minLength: 8,
            requireSpecialChars: args.plan !== "basic",
            requireNumbers: true,
          },
          maxFailedLogins: 5,
          requireTwoFactor: args.plan === "enterprise",
        },
        customization: {
          primaryColor: "#2563eb",
          welcomeMessage: `Welcome to ${args.name}`,
          languages: ["en"],
        },
      },
      metadata: {
        createdAt: Date.now(),
        createdBy: args.creatorUserId,
        lastUpdated: Date.now(),
        timezone: "Africa/Nairobi",
        country: "Kenya",
        currency: "KES",
      },
      isActive: true,
    });

    // Assign creator as estate admin
    await ctx.db.insert("estate_users", {
      estateId,
      userId: args.creatorUserId,
      role: "estate_admin",
      permissions: ["*"], // Full permissions
      isActive: true,
      assignedAt: Date.now(),
      assignedBy: args.creatorUserId,
    });

    // Create subscription record
    const planPricing = {
      basic: 5000, // KES per month
      premium: 15000,
      enterprise: 35000,
    };

    await ctx.db.insert("estate_subscriptions", {
      estateId,
      plan: args.plan,
      status: "active",
      startDate: Date.now(),
      endDate: Date.now() + (365 * 24 * 60 * 60 * 1000), // 1 year
      billingAmount: planPricing[args.plan],
      currency: "KES",
      features: getFeaturesByPlan(args.plan),
      limits: getLimitsByPlan(args.plan),
      nextPayment: Date.now() + (30 * 24 * 60 * 60 * 1000),
    });

    // Log estate creation
    await ctx.db.insert("audit_logs", {
      userId: args.creatorUserId,
      action: "estate_created",
      entityType: "estate",
      entityId: estateId,
      details: {
        estateName: args.name,
        plan: args.plan,
      },
    });

    return { success: true, estateId };
  },
});

export const getEstateDetails = query({
  args: { estateId: v.id("estates") },
  handler: async (ctx, args) => {
    const estate = await ctx.db.get(args.estateId);
    if (!estate) return null;

    // Get subscription details
    const subscription = await ctx.db
      .query("estate_subscriptions")
      .withIndex("by_estate", (q) => q.eq("estateId", args.estateId))
      .filter((q) => q.eq(q.field("status"), "active"))
      .first();

    // Get user count
    const userCount = await ctx.db
      .query("estate_users")
      .withIndex("by_estate", (q) => q.eq("estateId", args.estateId))
      .filter((q) => q.eq(q.field("isActive"), true))
      .collect()
      .then(users => users.length);

    // Get unit count
    const unitCount = await ctx.db
      .query("estate_units")
      .withIndex("by_estate", (q) => q.eq("estateId", args.estateId))
      .collect()
      .then(units => units.length);

    return {
      ...estate,
      subscription,
      stats: {
        userCount,
        unitCount,
        occupiedUnits: 0, // TODO: Calculate
      },
    };
  },
});

export const updateEstateSettings = mutation({
  args: {
    estateId: v.id("estates"),
    updates: v.object({
      settings: v.optional(v.any()),
      contact: v.optional(v.any()),
    }),
    updatedBy: v.id("users"),
  },
  handler: async (ctx, args) => {
    const estate = await ctx.db.get(args.estateId);
    if (!estate) {
      throw new Error("Estate not found");
    }

    const updateData: any = {
      metadata: {
        ...estate.metadata,
        lastUpdated: Date.now(),
      },
    };

    if (args.updates.settings) {
      updateData.settings = { ...estate.settings, ...args.updates.settings };
    }

    if (args.updates.contact) {
      updateData.contact = { ...estate.contact, ...args.updates.contact };
    }

    await ctx.db.patch(args.estateId, updateData);

    // Log settings update
    await ctx.db.insert("audit_logs", {
      userId: args.updatedBy,
      action: "estate_settings_updated",
      entityType: "estate",
      entityId: args.estateId,
      details: args.updates,
    });

    return { success: true };
  },
});

// Helper functions
function getFeaturesByPlan(plan: string): string[] {
  const features = {
    basic: ["visitor_management", "basic_reports"],
    premium: ["visitor_management", "sms_notifications", "advanced_reports", "analytics"],
    enterprise: ["visitor_management", "sms_notifications", "advanced_reports", "analytics", "biometric_entry", "api_access"],
  };
  return features[plan] || features.basic;
}

function getLimitsByPlan(plan: string) {
  const limits = {
    basic: { maxUsers: 50, maxVisits: 500, maxStorage: 1000 },
    premium: { maxUsers: 200, maxVisits: 2000, maxStorage: 5000 },
    enterprise: { maxUsers: 1000, maxVisits: 10000, maxStorage: 20000 },
  };
  return limits[plan] || limits.basic;
}
```

```typescript
  Building
} from 'lucide-react';
import { api } from '@/convex/_generated/api';
import { Id } from '@/convex/_generated/dataModel';

interface EstateSettingsPanelProps {
  estateId: Id<"estates">;
  userId: Id<"users">;
}

export function EstateSettingsPanel({ estateId, userId }: EstateSettingsPanelProps) {
  const [activeTab, setActiveTab] = useState("general");
  const [hasChanges, setHasChanges] = useState(false);

  const estate = useQuery(api.estateManagement.getEstateDetails, { estateId });
  const updateSettings = useMutation(api.estateManagement.updateEstateSettings);

  const [settings, setSettings] = useState(estate?.settings || {});

  const handleSettingChange = (category: string, field: string, value: any) => {
    setSettings(prev => ({
      ...prev,
      [category]: {
        ...prev[category],
        [field]: value,
      },
    }));
    setHasChanges(true);
  };

  const handleSave = async () => {
    try {
      await updateSettings({
        estateId,
        updates: { settings },
        updatedBy: userId,
      });
      setHasChanges(false);
    } catch (error) {
      console.error("Failed to update settings:", error);
    }
  };

  if (!estate) {
    return <div>Loading estate settings...</div>;
  }

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex justify-between items-center">
        <div>
          <h2 className="text-2xl font-bold">{estate.name} Settings</h2>
          <p className="text-gray-600">Configure estate-specific settings and preferences</p>
        </div>
        <Button onClick={handleSave} disabled={!hasChanges}>
          Save Changes
        </Button>
      </div>

      <Card>
        <CardContent className="p-0">
          <Tabs value={activeTab} onValueChange={setActiveTab}>
            <TabsList className="grid w-full grid-cols-6">
              <TabsTrigger value="general">
                <Settings className="w-4 h-4 mr-2" />
                General
              </TabsTrigger>
              <TabsTrigger value="security">
                <Shield className="w-4 h-4 mr-2" />
                Security
              </TabsTrigger>
              <TabsTrigger value="features">
                <Building className="w-4 h-4 mr-2" />
                Features
              </TabsTrigger>
              <TabsTrigger value="notifications">
                <Bell className="w-4 h-4 mr-2" />
                Notifications
              </TabsTrigger>
              <TabsTrigger value="appearance">
                <Palette className="w-4 h-4 mr-2" />
                Appearance
              </TabsTrigger>
              <TabsTrigger value="hours">
                <Clock className="w-4 h-4 mr-2" />
                Hours
              </TabsTrigger>
            </TabsList>

            <div className="p-6">
              <TabsContent value="general" className="space-y-6">
                <div>
                  <h3 className="text-lg font-semibold mb-4">Visitor Management</h3>
                  <div className="space-y-4">
                    <div className="flex items-center justify-between">
                      <div>
                        <Label>Allow Walk-in Visitors</Label>
                        <p className="text-sm text-gray-600">Permit guests without pre-registration</p>
                      </div>
                      <Switch
                        checked={settings.allowWalkIns}
                        onCheckedChange={(checked) => 
                          handleSettingChange('general', 'allowWalkIns', checked)
                        }
                      />
                    </div>

                    <div className="flex items-center justify-between">
                      <div>
                        <Label>Require Photo Upload</Label>
                        <p className="text-sm text-gray-600">Mandatory photo for all visitors</p>
                      </div>
                      <Switch
                        checked={settings.requirePhotoUpload}
                        onCheckedChange={(checked) => 
                          handleSettingChange('general', 'requirePhotoUpload', checked)
                        }
                      />
                    </div>

                    <div>
                      <Label htmlFor="overstayThreshold">Overstay Threshold (hours)</Label>
                      <Input
                        id="overstayThreshold"
                        type="number"
                        value={settings.overstayThresholdHours}
                        onChange={(e) => 
                          handleSettingChange('general', 'overstayThresholdHours', parseInt(e.target.value))
                        }
                        className="w-32"
                      />
                    </div>
                  </div>
                </div>
              </TabsContent>

              <TabsContent value="security" className="space-y-6">
                <div>
                  <h3 className="text-lg font-semibold mb-4">Security Policies</h3>
                  <div className="space-y-4">
                    <div>
                      <Label htmlFor="sessionTimeout">Session Timeout (minutes)</Label>
                      <Input
                        id="sessionTimeout"
                        type="number"
                        value={Math.floor(settings.security?.sessionTimeout / (60 * 1000)) || 30}
                        onChange={(e) => 
                          handleSettingChange('security', 'sessionTimeout', parseInt(e.target.value) * 60 * 1000)
                        }
                        className="w-32"
                      />
                    </div>

                    <div>
                      <Label htmlFor="maxFailedLogins">Max Failed Login Attempts</Label>
                      <Input
                        id="maxFailedLogins"
                        type="number"
                        value={settings.security?.maxFailedLogins || 5}
                        onChange={(e) => 
                          handleSettingChange('security', 'maxFailedLogins', parseInt(e.target.value))
                        }
                        className="w-32"
                      />
                    </div>

                    <div className="flex items-center justify-between">
                      <div>
                        <Label>Require Two-Factor Authentication</Label>
                        <p className="text-sm text-gray-600">Mandatory 2FA for all users</p>
                      </div>
                      <Switch
                        checked={settings.security?.requireTwoFactor || false}
                        onCheckedChange={(checked) => 
                          handleSettingChange('security', 'requireTwoFactor', checked)
                        }
                      />
                    </div>
                  </div>
                </div>
              </TabsContent>

              <TabsContent value="features" className="space-y-6">
                <div>
                  <h3 className="text-lg font-semibold mb-4">Available Features</h3>
                  <div className="grid grid-cols-2 gap-4">
                    {Object.entries(settings.features || {}).map(([feature, enabled]) => (
                      <div key={feature} className="flex items-center justify-between p-3 border rounded">
                        <div>
                          <Label className="capitalize">
                            {feature.replace(/([A-Z])/g, ' $1').trim()}
                          </Label>
                        </div>
                        <Switch
                          checked={enabled}
                          onCheckedChange={(checked) => 
                            handleSettingChange('features', feature, checked)
                          }
                        />
                      </div>
                    ))}
                  </div>
                </div>
              </TabsContent>

              <TabsContent value="notifications" className="space-y-6">
                <div>
                  <h3 className="text-lg font-semibold mb-4">Notification Settings</h3>
                  <div className="space-y-4">
                    <div className="flex items-center justify-between">
                      <div>
                        <Label>SMS Notifications</Label>
                        <p className="text-sm text-gray-600">Send SMS alerts and updates</p>
                      </div>
                      <Switch
                        checked={settings.features?.smsNotifications || false}
                        onCheckedChange={(checked) => 
                          handleSettingChange('features', 'smsNotifications', checked)
                        }
                      />
                    </div>

                    <div className="flex items-center justify-between">
                      <div>
                        <Label>Email Notifications</Label>
                        <p className="text-sm text-gray-600">Send email alerts and reports</p>
                      </div>
                      <Switch
                        checked={settings.features?.emailNotifications || false}
                        onCheckedChange={(checked) => 
                          handleSettingChange('features', 'emailNotifications', checked)
                        }
                      />
                    </div>
                  </div>
                </div>
              </TabsContent>

              <TabsContent value="appearance" className="space-y-6">
                <div>
                  <h3 className="text-lg font-semibold mb-4">Customization</h3>
                  <div className="space-y-4">
                    <div>
                      <Label htmlFor="primaryColor">Primary Color</Label>
                      <Input
                        id="primaryColor"
                        type="color"
                        value={settings.customization?.primaryColor || "#2563eb"}
                        onChange={(e) => 
                          handleSettingChange('customization', 'primaryColor', e.target.value)
                        }
                        className="w-20 h-10"
                      />
                    </div>

                    <div>
                      <Label htmlFor="welcomeMessage">Welcome Message</Label>
                      <Input
                        id="welcomeMessage"
                        value={settings.customization?.welcomeMessage || ""}
                        onChange={(e) => 
                          handleSettingChange('customization', 'welcomeMessage', e.target.value)
                        }
                      />
                    </div>
                  </div>
                </div>
              </TabsContent>

              <TabsContent value="hours" className="space-y-6">
                <div>
                  <h3 className="text-lg font-semibold mb-4">Operating Hours</h3>
                  <div className="grid grid-cols-2 gap-4">
                    <div>
                      <Label htmlFor="startTime">Start Time</Label>
                      <Input
                        id="startTime"
                        type="time"
                        value={settings.workingHours?.start || "09:00"}
                        onChange={(e) => 
                          handleSettingChange('workingHours', 'start', e.target.value)
                        }
                      />
                    </div>
                    <div>
                      <Label htmlFor="endTime">End Time</Label>
                      <Input
                        id="endTime"
                        type="time"
                        value={settings.workingHours?.end || "17:00"}
                        onChange={(e) => 
                          handleSettingChange('workingHours', 'end', e.target.value)
                        }
                      />
                    </div>
                  </div>
                </div>
              </TabsContent>
            </div>
          </Tabs>
        </CardContent>
      </Card>
    </div>
  );
}
```
