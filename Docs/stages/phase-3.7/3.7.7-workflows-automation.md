// convex/automation/workflowEngine.ts
import { action, mutation, query } from "../_generated/server";
import { v } from "convex/values";

export const triggerWorkflow = action({
  args: {
    triggerType: v.string(),
    estateId: v.id("estates"),
    data: v.any(),
    entityId: v.optional(v.string()),
    userId: v.optional(v.id("users")),
  },
  handler: async (ctx, args) => {
    // Find matching workflows
    const workflows = await ctx.runQuery("automation/workflowEngine.getActiveWorkflows", {
      estateId: args.estateId,
      triggerType: args.triggerType,
    });

    const executionIds = [];

    for (const workflow of workflows) {
      // Evaluate trigger conditions
      if (await evaluateConditions(workflow.trigger.conditions, args.data)) {
        // Start workflow execution
        const executionId = await ctx.runMutation("automation/workflowEngine.startExecution", {
          workflowId: workflow._id,
          estateId: args.estateId,
          triggeredBy: {
            type: args.triggerType,
            entityId: args.entityId,
            userId: args.userId,
            data: args.data,
          },
        });

        executionIds.push(executionId);

        // Execute workflow asynchronously
        ctx.scheduler.runAfter(0, "automation/workflowEngine.executeWorkflow", {
          executionId,
        });
      }
    }

    return { triggered: executionIds.length, executionIds };
  },
});

export const getActiveWorkflows = query({
  args: {
    estateId: v.id("estates"),
    triggerType: v.string(),
  },
  handler: async (ctx, args) => {
    return await ctx.db
      .query("workflows")
      .withIndex("by_estate", (q) => q.eq("estateId", args.estateId))
      .filter((q) => 
        q.and(
          q.eq(q.field("isActive"), true),
          q.eq(q.field("trigger.type"), args.triggerType)
        )
      )
      .collect();
  },
});

export const startExecution = mutation({
  args: {
    workflowId: v.id("workflows"),
    estateId: v.id("estates"),
    triggeredBy: v.any(),
  },
  handler: async (ctx, args) => {
    const executionId = await ctx.db.insert("workflow_executions", {
      workflowId: args.workflowId,
      estateId: args.estateId,
      triggeredBy: args.triggeredBy,
      status: "pending",
      startedAt: Date.now(),
      currentActionIndex: 0,
      results: [],
    });

    return executionId;
  },
});

export const executeWorkflow = action({
  args: { executionId: v.id("workflow_executions") },
  handler: async (ctx, args) => {
    const execution = await ctx.runQuery("workflow_executions", { id: args.executionId });
    if (!execution) return;

    const workflow = await ctx.runQuery("workflows", { id: execution.workflowId });
    if (!workflow) return;

    try {
      // Update execution status
      await ctx.runMutation("automation/workflowEngine.updateExecutionStatus", {
        executionId: args.executionId,
        status: "running",
      });

      // Execute actions sequentially
      for (let i = execution.currentActionIndex; i < workflow.actions.length; i++) {
        const action = workflow.actions[i];
        
        // Apply delay if specified
        if (action.delay && action.delay > 0) {
          ctx.scheduler.runAfter(action.delay, "automation/workflowEngine.executeAction", {
            executionId: args.executionId,
            actionIndex: i,
          });
          return; // Continue execution after delay
        }

        const result = await executeAction(ctx, action, execution.triggeredBy.data, execution.estateId);
        
        await ctx.runMutation("automation/workflowEngine.recordActionResult", {
          executionId: args.executionId,
          actionIndex: i,
          result,
        });

        // Update current action index
        await ctx.runMutation("automation/workflowEngine.updateCurrentAction", {
          executionId: args.executionId,
          actionIndex: i + 1,
        });
      }

      // Complete execution
      await ctx.runMutation("automation/workflowEngine.updateExecutionStatus", {
        executionId: args.executionId,
        status: "completed",
        completedAt: Date.now(),
      });

    } catch (error) {
      await ctx.runMutation("automation/workflowEngine.updateExecutionStatus", {
        executionId: args.executionId,
        status: "failed",
        error: error.message,
        completedAt: Date.now(),
      });
    }
  },
});

export const executeAction = action({
  args: {
    executionId: v.id("workflow_executions"),
    actionIndex: v.number(),
  },
  handler: async (ctx, args) => {
    const execution = await ctx.runQuery("workflow_executions", { id: args.executionId });
    if (!execution) return;

    const workflow = await ctx.runQuery("workflows", { id: execution.workflowId });
    if (!workflow) return;

    const action = workflow.actions[args.actionIndex];
    const result = await executeAction(ctx, action, execution.triggeredBy.data, execution.estateId);
    
    await ctx.runMutation("automation/workflowEngine.recordActionResult", {
      executionId: args.executionId,
      actionIndex: args.actionIndex,
      result,
    });

    // Continue workflow execution
    if (args.actionIndex + 1 < workflow.actions.length) {
      ctx.scheduler.runAfter(0, "automation/workflowEngine.executeAction", {
        executionId: args.executionId,
        actionIndex: args.actionIndex + 1,
      });
    } else {
      // Complete execution
      await ctx.runMutation("automation/workflowEngine.updateExecutionStatus", {
        executionId: args.executionId,
        status: "completed",
        completedAt: Date.now(),
      });
    }
  },
});

// Helper functions
async function evaluateConditions(conditions: any[], data: any): Promise<boolean> {
  for (const condition of conditions) {
    const fieldValue = getFieldValue(data, condition.field);
    const conditionValue = condition.value;

    let result = false;
    switch (condition.operator) {
      case "equals":
        result = fieldValue === conditionValue;
        break;
      case "not_equals":
        result = fieldValue !== conditionValue;
        break;
      case "greater_than":
        result = fieldValue > conditionValue;
        break;
      case "less_than":
        result = fieldValue < conditionValue;
        break;
      case "contains":
        result = String(fieldValue).includes(String(conditionValue));
        break;
      case "starts_with":
        result = String(fieldValue).startsWith(String(conditionValue));
        break;
      case "in":
        result = Array.isArray(conditionValue) && conditionValue.includes(fieldValue);
        break;
      case "exists":
        result = fieldValue !== undefined && fieldValue !== null;
        break;
    }

    if (!result) return false; // All conditions must be true (AND logic)
  }
  return true;
}

async function executeAction(ctx: any, action: any, triggerData: any, estateId: string) {
  switch (action.type) {
    case "send_notification":
      return await sendNotification(ctx, action.configuration, triggerData, estateId);
    case "send_email":
      return await sendEmail(ctx, action.configuration, triggerData, estateId);
    case "send_sms":
      return await sendSMS(ctx, action.configuration, triggerData, estateId);
    case "update_record":
      return await updateRecord(ctx, action.configuration, triggerData);
    case "create_record":
      return await createRecord(ctx, action.configuration, triggerData);
    case "call_webhook":
      return await callWebhook(ctx, action.configuration, triggerData);
    case "generate_report":
      return await generateReport(ctx, action.configuration, estateId);
    case "escalate_incident":
      return await escalateIncident(ctx, action.configuration, triggerData, estateId);
    case "auto_checkout":
      return await autoCheckout(ctx, action.configuration, triggerData);
    case "block_visitor":
      return await blockVisitor(ctx, action.configuration, triggerData);
    default:
      throw new Error(`Unknown action type: ${action.type}`);
  }
}

// Action implementations
async function sendNotification(ctx: any, config: any, data: any, estateId: string) {
  const message = interpolateTemplate(config.message, data);
  
  return await ctx.runMutation("notifications.sendNotification", {
    estateId,
    recipients: config.recipients,
    title: config.title,
    message,
    type: config.notificationType || "info",
  });
}

async function sendEmail(ctx: any, config: any, data: any, estateId: string) {
  const subject = interpolateTemplate(config.subject, data);
  const body = interpolateTemplate(config.body, data);

  return await ctx.runAction("integrations.sendEmail", {
    estateId,
    to: config.recipients,
    subject,
    body,
    template: config.template,
  });
}

async function sendSMS(ctx: any, config: any, data: any, estateId: string) {
  const message = interpolateTemplate(config.message, data);

  return await ctx.runAction("integrations.sendSMS", {
    estateId,
    to: config.recipients,
    message,
    type: "automation",
  });
}

function getFieldValue(data: any, fieldPath: string): any {
  return fieldPath.split('.').reduce((obj, key) => obj?.[key], data);
}

function interpolateTemplate(template: string, data: any): string {
  return template.replace(/\{\{([^}]+)\}\}/g, (match, fieldPath) => {
    const value = getFieldValue(data, fieldPath.trim());
    return value !== undefined ? String(value) : match;
  });
}

// Helper functions
async function evaluateRuleConditions(conditions: any[], data: any): Promise<boolean> {
  let result = true;
  let currentLogicalOperator = "AND";

  for (const condition of conditions) {
    const conditionResult = evaluateSingleCondition(condition, data);
    
    if (currentLogicalOperator === "AND") {
      result = result && conditionResult;
    } else if (currentLogicalOperator === "OR") {
      result = result || conditionResult;
    }

    if (condition.logicalOperator) {
      currentLogicalOperator = condition.logicalOperator;
    }
  }

  return result;
}

function evaluateSingleCondition(condition: any, data: any): boolean {
  const fieldValue = getFieldValue(data, condition.field);
  const conditionValue = condition.value;

  switch (condition.operator) {
    case "equals":
      return fieldValue === conditionValue;
    case "not_equals":
      return fieldValue !== conditionValue;
    case "greater_than":
      return Number(fieldValue) > Number(conditionValue);
    case "less_than":
      return Number(fieldValue) < Number(conditionValue);
    case "contains":
      return String(fieldValue).includes(String(conditionValue));
    case "in":
      return Array.isArray(conditionValue) && conditionValue.includes(fieldValue);
    case "exists":
      return fieldValue !== undefined && fieldValue !== null;
    default:
      return false;
  }
}

async function executeRuleActions(ctx: any, actions: any[], data: any, estateId: string) {
  const results = [];
  
  for (const action of actions) {
    try {
      const result = await executeAction(ctx, action, data, estateId);
      results.push({ action: action.type, success: true, result });
    } catch (error) {
      results.push({ action: action.type, success: false, error: error.message });
    }
  }

  return results;
}

function getFieldValue(data: any, fieldPath: string): any {
  return fieldPath.split('.').reduce((obj, key) => obj?.[key], data);
}

      for (const channel of userChannels) {

        );

        // Send notification

export const scheduleSmartReminders = action({
  args: { estateId: v.id("estates") },
  handler: async (ctx, args) => {
    const currentTime = Date.now();
    
    // Check for upcoming appointments/visits

      estateId: args.estateId,

    });

      if (shouldSendReminder(visit, currentTime)) {
        const notificationId = await ctx.runAction("automation/smartNotifications.sendSmartNotification", {
          estateId: args.estateId,
          event: "upcoming_visit_reminder",
          data: visit,
          specificRecipients: [visit.residentId],
          priority: "medium",
        });
        notifications.push(notificationId);
      }
    }

    // Process overdue checkouts
    for (const checkout of overdueCheckouts) {
      const notificationId = await ctx.runAction("automation/smartNotifications.sendSmartNotification", {
        estateId: args.estateId,
        event: "overdue_checkout_alert",
        data: checkout,
        recipientRoles: ["admin", "guard"],
        priority: "high",
      });
      notifications.push(notificationId);

        event: "invitation_expiring",
        data: invitation,

function determineNotificationChannels(priority: string): string[] {
  switch (priority) {
    case "urgent":
      return ["push", "sms", "email", "in_app"];
    case "high":
      return ["push", "email", "in_app"];
    case "medium":
      return ["push", "in_app"];
    case "low":
      return ["in_app"];
    default:
      return ["in_app"];
  }

async function getNotificationRecipients(
  ctx: any, 
  estateId: string, 
  roles?: string[], 
  specificUsers?: string[]
) {
  const recipients = [];

  if (specificUsers) {
    for (const userId of specificUsers) {
      const user = await ctx.runQuery("users.getUser", { userId });
      if (user) recipients.push(user);
    }
  }

  if (roles) {
    for (const role of roles) {
      const roleUsers = await ctx.runQuery("users.getUsersByRole", { 
        estateId, 
        role 
      });
      recipients.push(...roleUsers);
    }
  }

  return recipients;
}

async function generatePersonalizedMessage(
  ctx: any, 
  event: string, 
  data: any, 
  recipient: any, 
  channel: string
) {
  // Get template for event and channel
  const template = await ctx.runQuery("automation/smartNotifications.getTemplate", {
    estateId: recipient.estateId,
    event,
    channel,
    language: recipient.preferredLanguage || "en",
  });

  if (!template) {
    return getDefaultMessage(event, data, channel);
  }

  // Interpolate template variables
  return interpolateTemplate(template.template, {
    ...data,
    recipientName: recipient.name,
    recipientRole: recipient.role,
  });
}

function shouldSendReminder(visit: any, currentTime: number): boolean {
  const timeDiff = visit.scheduledTime - currentTime;
  const hoursDiff = timeDiff / (1000 * 60 * 60);
  
  // Send reminder 24 hours before, 2 hours before, and 30 minutes before
  return hoursDiff <= 24 && hoursDiff >= 23.5 ||
         hoursDiff <= 2 && hoursDiff >= 1.5 ||
         hoursDiff <= 0.5 && hoursDiff >= 0;
}

function interpolateTemplate(template: string, data: any): string {
  return template.replace(/\{\{([^}]+)\}\}/g, (match, key) => {
    return data[key] || match;
  });
}
