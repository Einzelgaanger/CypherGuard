```typescript
// convex/api/authentication.ts
import { mutation, query } from "../_generated/server";
import { v } from "convex/values";
import crypto from "crypto";

export const createAPIKey = mutation({
  args: {
    name: v.string(),
    description: v.string(),
    estateId: v.optional(v.id("estates")),
    permissions: v.array(v.string()),
    rateLimit: v.object({
      requestsPerMinute: v.number(),
      requestsPerHour: v.number(),
      requestsPerDay: v.number(),
    }),
    expiresIn: v.optional(v.number()), // days
    createdBy: v.id("users"),
  },
  handler: async (ctx, args) => {
    // Generate API key
    const keyId = generateKeyId();
    const rawKey = generateRawAPIKey();
    const hashedKey = hashAPIKey(rawKey);

    // Calculate expiration
    const expiresAt = args.expiresIn ? 
      Date.now() + (args.expiresIn * 24 * 60 * 60 * 1000) : 
      undefined;

    // Store API key
    const apiKeyId = await ctx.db.insert("api_keys", {
      keyId,
      hashedKey,
      name: args.name,
      description: args.description,
      estateId: args.estateId,
      permissions: args.permissions,
      rateLimit: args.rateLimit,
      isActive: true,
      createdAt: Date.now(),
      createdBy: args.createdBy,
      expiresAt,
    });

    // Log API key creation
    await ctx.db.insert("audit_logs", {
      userId: args.createdBy,
      action: "api_key_created",
      entityType: "api_key",
      entityId: apiKeyId,
      details: {
        keyName: args.name,
        permissions: args.permissions,
        estateId: args.estateId,
      },
    });

    return {
      success: true,
      keyId,
      key: `${keyId}.${rawKey}`, // Return full key only once
      expiresAt,
    };
  },
});

export const validateAPIKey = query({
  args: { 
    key: v.string(),
    requiredPermission: v.optional(v.string()),
    estateId: v.optional(v.id("estates")),
  },
  handler: async (ctx, args) => {
    try {
      // Parse key
      const [keyId, rawKey] = args.key.split('.');
      if (!keyId || !rawKey) {
        return { valid: false, error: "Invalid key format" };
      }

      // Find API key
      const apiKey = await ctx.db
        .query("api_keys")
        .withIndex("by_key_id", (q) => q.eq("keyId", keyId))
        .first();

      if (!apiKey) {
        return { valid: false, error: "API key not found" };
      }

      // Check if key is active
      if (!apiKey.isActive) {
        return { valid: false, error: "API key is disabled" };
      }

      // Check expiration
      if (apiKey.expiresAt && apiKey.expiresAt < Date.now()) {
        return { valid: false, error: "API key has expired" };
      }

      // Verify key hash
      const providedHash = hashAPIKey(rawKey);
      if (providedHash !== apiKey.hashedKey) {
        return { valid: false, error: "Invalid API key" };
      }

      // Check estate access
      if (args.estateId && apiKey.estateId && apiKey.estateId !== args.estateId) {
        return { valid: false, error: "Estate access denied" };
      }

      // Check permissions
      if (args.requiredPermission && !apiKey.permissions.includes(args.requiredPermission) && !apiKey.permissions.includes("*")) {
        return { valid: false, error: "Insufficient permissions" };
      }

      // Check rate limits
      const rateLimitResult = await checkRateLimit(ctx, keyId, apiKey.rateLimit);
      if (!rateLimitResult.allowed) {
        return { 
          valid: false, 
          error: "Rate limit exceeded",
          retryAfter: rateLimitResult.retryAfter 
        };
      }

      // Update last used
      await ctx.db.patch(apiKey._id, { lastUsed: Date.now() });

      return {
        valid: true,
        keyId,
        permissions: apiKey.permissions,
        estateId: apiKey.estateId,
        rateLimit: rateLimitResult.remaining,
      };
    } catch (error) {
      return { valid: false, error: "Validation failed" };
    }
  },
});

export const revokeAPIKey = mutation({
  args: {
    keyId: v.string(),
    revokedBy: v.id("users"),
  },
  handler: async (ctx, args) => {
    const apiKey = await ctx.db
      .query("api_keys")
      .withIndex("by_key_id", (q) => q.eq("keyId", args.keyId))
      .first();

    if (!apiKey) {
      throw new Error("API key not found");
    }

    await ctx.db.patch(apiKey._id, { isActive: false });

    // Log revocation
    await ctx.db.insert("audit_logs", {
      userId: args.revokedBy,
      action: "api_key_revoked",
      entityType: "api_key",
      entityId: apiKey._id,
      details: {
        keyName: apiKey.name,
        keyId: args.keyId,
      },
    });

    return { success: true };
  },
});

// Helper functions
function generateKeyId(): string {
  return `ck_${crypto.randomBytes(8).toString('hex')}`;
}

function generateRawAPIKey(): string {
  return crypto.randomBytes(32).toString('hex');
}

function hashAPIKey(rawKey: string): string {
  return crypto.createHash('sha256').update(rawKey).digest('hex');
}

async function checkRateLimit(ctx: any, keyId: string, limits: any) {
  const now = Date.now();
  const buckets = ['minute', 'hour', 'day'];
  const periods = {
    minute: 60 * 1000,
    hour: 60 * 60 * 1000,
    day: 24 * 60 * 60 * 1000,
  };
  const maxRequests = {
    minute: limits.requestsPerMinute,
    hour: limits.requestsPerHour,
    day: limits.requestsPerDay,
  };

  for (const bucket of buckets) {
    const period = periods[bucket];
    const max = maxRequests[bucket];
    const resetAt = Math.ceil(now / period) * period;

    // Get or create bucket
    let bucketDoc = await ctx.db
      .query("rate_limit_buckets")
      .withIndex("by_key_bucket", (q) => q.eq("keyId", keyId).eq("bucket", bucket))
      .first();

    if (!bucketDoc || bucketDoc.resetAt <= now) {
      // Create or reset bucket
      if (bucketDoc) {
        await ctx.db.patch(bucketDoc._id, { count: 1, resetAt });
      } else {
        await ctx.db.insert("rate_limit_buckets", {
          keyId,
          bucket,
          count: 1,
          resetAt,
        });
      }
    } else {
      // Check if limit exceeded
      if (bucketDoc.count >= max) {
        return {
          allowed: false,
          retryAfter: bucketDoc.resetAt - now,
        };
      }

      // Increment counter
      await ctx.db.patch(bucketDoc._id, { count: bucketDoc.count + 1 });
    }
  }

  return { allowed: true, remaining: limits.requestsPerMinute };
}
```

```typescript
// Generic API request handler
async function handleAPIRequest(
  ctx: any, 
  request: Request, 
  requiredPermission: string,
  handler: (auth: any, params?: any, body?: any) => Promise<any>
) {
  try {
    const startTime = Date.now();
    
    // Extract API key from Authorization header
    const authHeader = request.headers.get("Authorization");
    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return new Response(JSON.stringify({ error: "Missing or invalid authorization header" }), {
        status: 401,
        headers: { "Content-Type": "application/json" },
      });
    }

    const apiKey = authHeader.substring(7);
    
    // Validate API key
    const auth = await ctx.runQuery(api.api.authentication.validateAPIKey, {
      key: apiKey,
      requiredPermission,
    });

    if (!auth.valid) {
      await logAPIRequest(ctx, null, request, 401, auth.error);
      return new Response(JSON.stringify({ error: auth.error }), {
        status: auth.error === "Rate limit exceeded" ? 429 : 401,
        headers: { 
          "Content-Type": "application/json",
          ...(auth.retryAfter ? { "Retry-After": String(Math.ceil(auth.retryAfter / 1000)) } : {}),
        },
      });
    }

    // Parse URL parameters
    const url = new URL(request.url);
    const params = Object.fromEntries(url.searchParams);
    
    // Parse request body
    let body = null;
    if (request.method !== "GET" && request.method !== "DELETE") {
      try {
        body = await request.json();
      } catch (error) {
        return new Response(JSON.stringify({ error: "Invalid JSON body" }), {
          status: 400,
          headers: { "Content-Type": "application/json" },
        });
      }
    }

    // Call handler
    const result = await handler(auth, params, body);
    const responseTime = Date.now() - startTime;
    
    // Log successful request
    await logAPIRequest(ctx, auth.keyId, request, result.status || 200, null, responseTime);

    return new Response(JSON.stringify(result.data || result), {
      status: result.status || 200,
      headers: { "Content-Type": "application/json" },
    });

  } catch (error) {
    const responseTime = Date.now() - Date.now();
    await logAPIRequest(ctx, null, request, 500, error.message, responseTime);
    
    return new Response(JSON.stringify({ error: "Internal server error" }), {
      status: 500,
      headers: { "Content-Type": "application/json" },
    });
  }
}

async function logAPIRequest(
  ctx: any,
  keyId: string | null,
  request: Request,
  statusCode: number,
  error?: string,
  responseTime?: number
) {
  const url = new URL(request.url);
  
  await ctx.runMutation(api.api.logging.logAPIRequest, {
    keyId: keyId || "unknown",
    endpoint: url.pathname,
    method: request.method as any,
    statusCode,
    responseTime: responseTime || 0,
    requestSize: parseInt(request.headers.get("content-length") || "0"),
    responseSize: 0, // Calculate based on response
    ipAddress: request.headers.get("x-forwarded-for") || "unknown",
    userAgent: request.headers.get("user-agent"),
    error,
  });
}
```

```typescript
export const triggerWebhook = action({
  args: {
    event: v.string(),
    payload: v.any(),
    estateId: v.optional(v.id("estates")),
  },
  handler: async (ctx, args) => {
    // Find webhooks subscribed to this event
    let webhooksQuery = ctx.runQuery("webhooks", {});
    
    if (args.estateId) {
      webhooksQuery = ctx.runQuery("api.webhooks.getWebhooksByEstate", {
        estateId: args.estateId,
        event: args.event,
      });
    }

    const webhooks = await webhooksQuery;

    // Trigger each webhook
    for (const webhook of webhooks) {
      if (webhook.isActive && webhook.events.includes(args.event)) {
        await deliverWebhook(ctx, webhook, args.event, args.payload);
      }
    }

    return { triggered: webhooks.length };
  },
});

export const getWebhooksByEstate = query({
  args: { 
    estateId: v.id("estates"),
    event: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    let query = ctx.db
      .query("webhooks")
      .withIndex("by_estate", (q) => q.eq("estateId", args.estateId));

    const webhooks = await query.collect();

    if (args.event) {
      return webhooks.filter(w => w.events.includes(args.event));
    }

    return webhooks;
  },
});

// Webhook delivery function
async function deliverWebhook(ctx: any, webhook: any, event: string, payload: any) {
  const deliveryId = await ctx.runMutation("api.webhooks.createDelivery", {
    webhookId: webhook._id,
    event,
    payload,
    deliveryAttempt: 1,
  });

  try {
    // Prepare webhook payload
    const webhookPayload = {
      event,
      timestamp: Date.now(),
      data: payload,
    };

    // Generate signature
    const signature = generateWebhookSignature(webhook.secret, JSON.stringify(webhookPayload));

    // Prepare headers
    const headers = {
      'Content-Type': 'application/json',
      'User-Agent': 'CypherSec-Webhooks/1.0',
      'X-CypherSec-Event': event,
      'X-CypherSec-Signature': signature,
      'X-CypherSec-Delivery': deliveryId,
      ...webhook.headers,
    };

    // Send webhook
    const response = await fetch(webhook.url, {
      method: 'POST',
      headers,
      body: JSON.stringify(webhookPayload),
    });

    const responseBody = await response.text();

    if (response.ok) {
      // Mark as delivered
      await ctx.runMutation("api.webhooks.markDelivered", {
        deliveryId,
        responseCode: response.status,
        responseBody: responseBody.substring(0, 1000), // Limit stored response
      });
    } else {
      // Schedule retry
      await scheduleWebhookRetry(ctx, deliveryId, webhook, event, payload, 1);
    }

  } catch (error) {
    // Schedule retry
    await scheduleWebhookRetry(ctx, deliveryId, webhook, event, payload, 1, error.message);
  }
}

async function scheduleWebhookRetry(
  ctx: any,
  deliveryId: string,
  webhook: any,
  event: string,
  payload: any,
  attempt: number,
  error?: string
) {
  if (attempt >= webhook.retryPolicy.maxRetries) {
    // Mark as failed
    await ctx.runMutation("api.webhooks.markFailed", {
      deliveryId,
      error: error || "Max retries exceeded",
    });
    return;
  }

  // Calculate next retry time
  const delay = webhook.retryPolicy.retryDelay * Math.pow(2, attempt - 1); // Exponential backoff
  const nextRetryAt = Date.now() + delay;

  await ctx.runMutation("api.webhooks.scheduleRetry", {
    deliveryId,
    nextRetryAt,
    error,
  });

  // In a real implementation, you would schedule a cron job or use a queue
  // For now, we'll rely on a periodic retry processor
}

function generateWebhookSignature(secret: string, payload: string): string {
  return `sha256=${crypto.createHmac('sha256', secret).update(payload).digest('hex')}`;
}
```

```typescript
export const createIntegration = mutation({
  args: {
    estateId: v.id("estates"),
    type: v.union(
      v.literal("sms_provider"),
      v.literal("email_provider"),
      v.literal("access_control"),
      v.literal("camera_system"),
      v.literal("billing_system"),
      v.literal("property_management")
    ),
    provider: v.string(),
    configuration: v.any(),
    credentials: v.any(),
  },
  handler: async (ctx, args) => {
    // Encrypt credentials
    const encryptedCredentials = encryptData(JSON.stringify(args.credentials));

    const integrationId = await ctx.db.insert("integrations", {
      estateId: args.estateId,
      type: args.type,
      provider: args.provider,
      configuration: args.configuration,
      credentials: encryptedCredentials,
      isActive: true,
      createdAt: Date.now(),
      updatedAt: Date.now(),
    });

    // Test the integration
    const testResult = await testIntegration(ctx, integrationId);
    
    if (!testResult.success) {
      await ctx.db.patch(integrationId, { 
        isActive: false,
        syncStatus: "error",
      });
    }

    return { success: true, integrationId, testResult };
  },
});

export const syncIntegration = action({
  args: {
    integrationId: v.id("integrations"),
    syncType: v.optional(v.union(v.literal("full"), v.literal("incremental"))),
  },
  handler: async (ctx, args) => {
    const integration = await ctx.runQuery("integrations", { id: args.integrationId });
    if (!integration) {
      throw new Error("Integration not found");
    }

    try {
      await ctx.runMutation("integrations/index.updateSyncStatus", {
        integrationId: args.integrationId,
        status: "pending",
      });

      // Decrypt credentials
      const credentials = JSON.parse(decryptData(integration.credentials));

      // Perform sync based on integration type
      const syncResult = await performSync(integration, credentials, args.syncType || "incremental");

      await ctx.runMutation("integrations/index.updateSyncStatus", {
        integrationId: args.integrationId,
        status: "success",
        lastSync: Date.now(),
      });

      return syncResult;

    } catch (error) {
      await ctx.runMutation("integrations/index.updateSyncStatus", {
        integrationId: args.integrationId,
        status: "error",
      });
      throw error;
    }
  },
});

export const updateSyncStatus = mutation({
  args: {
    integrationId: v.id("integrations"),
    status: v.union(v.literal("success"), v.literal("error"), v.literal("pending")),
    lastSync: v.optional(v.number()),
  },
  handler: async (ctx, args) => {
    await ctx.db.patch(args.integrationId, {
      syncStatus: args.status,
      lastSync: args.lastSync,
      updatedAt: Date.now(),
    });
    return { success: true };
  },
});

  handler: async (ctx, args) => {

    if (!smsIntegration || !smsIntegration.isActive) {

    const credentials = JSON.parse(decryptData(smsIntegration.credentials));
    
    // Send SMS based on provider
    const result = await sendSMSViaProvider(
      smsIntegration.provider,
      credentials,
      args.to,
      args.message
    );

    // Log the SMS
    await ctx.runMutation("api.logging.logSMS", {
      estateId: args.estateId,
      to: args.to,
      message: args.message,
      provider: smsIntegration.provider,
      success: result.success,
      messageId: result.messageId,
      cost: result.cost,
    });

    return result;
  },
});
```
